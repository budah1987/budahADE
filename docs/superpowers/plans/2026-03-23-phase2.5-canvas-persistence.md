# Phase 2.5: Canvas State Persistence — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Persist canvas state to `.budahade/canvas.json` so tiles, positions, chat history, and viewport survive app quit, task switching, and relaunch.

**Architecture:** Add `Codable` to all canvas model types (`CanvasElement`, `ElementKind`, `TileType`, `FrameData`, `TextData`, `AgentRole`). A new `CanvasPersistence` enum handles debounced save (1s) and load. `PlanCanvasState.didMutate()` triggers save. `TaskState.enterPlanMode()` loads. Terminal tiles serialize as shells (no Ghostty state). Chat tiles serialize messages + role so conversations resume.

**Tech Stack:** Swift `Codable`, `JSONEncoder`/`JSONDecoder`, `DispatchWorkItem` for debounce, `NSApplication.willTerminateNotification` for quit save.

**Spec Reference:** `docs/superpowers/specs/2026-03-23-workflow-v2-overhaul-design.md` — Phase 2.5 (lines 224–267)

        let c = try decoder.container(keyedBy: CodingKeys.self)
        let axisStr = try c.decode(String.self, forKey: .axis)

```swift
// In AgentRole.swift
extension AgentRole: Codable {
    enum CodingKeys: String, CodingKey {
        case id, name, systemPrompt, defaultModel, colorHex
    }

    init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        self.id = try c.decode(String.self, forKey: .id)
        self.name = try c.decode(String.self, forKey: .name)
        self.systemPrompt = try c.decode(String.self, forKey: .systemPrompt)
        self.defaultModel = try c.decode(AgentModel.self, forKey: .defaultModel)
        let hex = try c.decode(UInt32.self, forKey: .colorHex)
        self.color = Color(hex: hex)
    }

    func encode(to encoder: Encoder) throws {
        var c = encoder.container(keyedBy: CodingKeys.self)
        try c.encode(id, forKey: .id)
        try c.encode(name, forKey: .name)
        try c.encode(systemPrompt, forKey: .systemPrompt)
        try c.encode(defaultModel, forKey: .defaultModel)
        // Convert Color back to hex — approximate via NSColor
        let nsColor = NSColor(self.color).usingColorSpace(.sRGB) ?? NSColor.gray
        let r = UInt32(nsColor.redComponent * 255) << 16
        let g = UInt32(nsColor.greenComponent * 255) << 8
        let b = UInt32(nsColor.blueComponent * 255)
        try c.encode(r | g | b, forKey: .colorHex)
    }
}
```

extension TileType: Codable {
    enum CodingKeys: String, CodingKey {
        case type, panelId, agent, path, url, sessionId, role
    }

    init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        let type = try c.decode(String.self, forKey: .type)
        switch type {
        case "terminal":
            let panelId = try c.decode(UUID.self, forKey: .panelId)
            let agent = try c.decode(AgentMode.self, forKey: .agent)
            self = .terminal(panelId: panelId, agent: agent)
        case "stickyNote":
            self = .stickyNote
        case "markdown":
            let path = try c.decode(String.self, forKey: .path)
            self = .markdown(path: path)
        case "image":
            let path = try c.decode(String.self, forKey: .path)
            self = .image(path: path)
        case "browser":
            let url = try c.decodeIfPresent(URL.self, forKey: .url)
            self = .browser(url: url)
        case "chatAgent":
            let sessionId = try c.decode(UUID.self, forKey: .sessionId)
            let role = try c.decode(AgentRole.self, forKey: .role)
            self = .chatAgent(sessionId: sessionId, role: role)
        default:
            throw DecodingError.dataCorrupted(.init(codingPath: [CodingKeys.type], debugDescription: "Unknown tile type: \(type)"))
        }
    }

    func encode(to encoder: Encoder) throws {
        var c = encoder.container(keyedBy: CodingKeys.self)
        switch self {
        case .terminal(let panelId, let agent):
            try c.encode("terminal", forKey: .type)
            try c.encode(panelId, forKey: .panelId)
            try c.encode(agent, forKey: .agent)
        case .stickyNote:
            try c.encode("stickyNote", forKey: .type)
        case .markdown(let path):
            try c.encode("markdown", forKey: .type)
            try c.encode(path, forKey: .path)
        case .image(let path):
            try c.encode("image", forKey: .type)
            try c.encode(path, forKey: .path)
        case .browser(let url):
            try c.encode("browser", forKey: .type)
            try c.encodeIfPresent(url, forKey: .url)
        case .chatAgent(let sessionId, let role):
            try c.encode("chatAgent", forKey: .type)
            try c.encode(sessionId, forKey: .sessionId)
            try c.encode(role, forKey: .role)

// BudahADE/Plan/CanvasPersistence.swift
import Foundation

// MARK: - Canvas Snapshot (serializable subset of PlanCanvasState)

struct CanvasSnapshot: Codable {
    let elements: [CanvasElement]
    let zoom: CGFloat
    let panOffsetWidth: CGFloat
    let panOffsetHeight: CGFloat
    let chatMessages: [UUID: [ChatMessage]]  // sessionId → messages
}

// MARK: - Canvas Persistence

enum CanvasPersistence {
    private static let filename = "canvas.json"

    static func save(_ snapshot: CanvasSnapshot, to worktreePath: String) throws {
        let dir = (worktreePath as NSString).appendingPathComponent(".budahade")
        try FileManager.default.createDirectory(
            atPath: dir, withIntermediateDirectories: true
        )
        let path = (dir as NSString).appendingPathComponent(filename)
        let encoder = JSONEncoder()
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
        let data = try encoder.encode(snapshot)
        try data.write(to: URL(fileURLWithPath: path))
    }

    static func load(from worktreePath: String) -> CanvasSnapshot? {
        let path = (worktreePath as NSString)
            .appendingPathComponent(".budahade/\(filename)")
        guard FileManager.default.fileExists(atPath: path),
              let data = try? Data(contentsOf: URL(fileURLWithPath: path)) else {
            return nil
        }
        return try? JSONDecoder().decode(CanvasSnapshot.self, from: data)
    }
}
```

// In PlanCanvasState.swift

// MARK: - Persistence

private var saveDebounce: DispatchWorkItem?

func snapshot() -> CanvasSnapshot {
    // Collect chat messages from all sessions
    var chatMessages: [UUID: [ChatMessage]] = [:]
    for (sessionId, session) in chatSessions {
        if !session.messages.isEmpty {
            chatMessages[sessionId] = session.messages
        }
    }
    return CanvasSnapshot(
        elements: elements,
        zoom: zoom,
        panOffsetWidth: panOffset.width,
        panOffsetHeight: panOffset.height,
        chatMessages: chatMessages
    )
}

func restore(from snapshot: CanvasSnapshot) {
    elements = snapshot.elements
    zoom = snapshot.zoom
    panOffset = CGSize(width: snapshot.panOffsetWidth, height: snapshot.panOffsetHeight)

    // Restore chat agent sessions from persisted messages
    for element in allTiles {
        if case .tile(.chatAgent(let sessionId, let role)) = element.kind {
            let session = chatManager.createSession(
                model: role.defaultModel,
                agentMode: AgentMode(rawValue: role.id),
                systemPrompt: role.systemPrompt,
                workingDirectory: worktreePath
            )
            // The session gets a new ID from createSession, but we need the original ID
            // to match the tile. Remove the auto-created one and insert with correct ID.
            chatManager.sessions.removeValue(forKey: session.id)

            let restoredSession = AgentSession(
                id: sessionId,
                model: role.defaultModel,
                agentMode: AgentMode(rawValue: role.id),
                systemPrompt: role.systemPrompt,
                workingDirectory: worktreePath
            )
            // Restore messages
            if let messages = snapshot.chatMessages[sessionId] {
                restoredSession.messages = messages
                restoredSession.totalInputTokens = messages.reduce(0) { $0 + $1.inputTokens }
                restoredSession.totalOutputTokens = messages.reduce(0) { $0 + $1.outputTokens }
            }
            chatManager.sessions[sessionId] = restoredSession
            chatSessions[sessionId] = restoredSession
        }
    }

    // Terminal tiles restore as empty shells — panels created on demand when visible
    // (Terminal panels are already created by addTile when the tile is interacted with)
}

func saveNow() {
    do {
        try CanvasPersistence.save(snapshot(), to: worktreePath)
    } catch {
        print("[CanvasPersistence] Save failed: \(error)")
    }
}

private func scheduleSave() {
    saveDebounce?.cancel()
    let work = DispatchWorkItem { [weak self] in
        Task { @MainActor in
            self?.saveNow()
        }
    }
    saveDebounce = work
    DispatchQueue.main.asyncAfter(deadline: .now() + 1.0, execute: work)
}
```

private func updateTerminalPanelId(oldId: UUID, newId: UUID) {
    for i in elements.indices {
        if case .tile(.terminal(let pid, let agent)) = elements[i].kind, pid == oldId {
            elements[i].kind = .tile(.terminal(panelId: newId, agent: agent))
            return
        }

                }
            }
        }
    }
